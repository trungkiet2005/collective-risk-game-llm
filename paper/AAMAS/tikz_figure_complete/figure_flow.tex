\newcommand{\fsize}[1]{\fontsize{#1}{\dimexpr #1pt*6/5\relax}\selectfont}
% #1 filename stem, #2 x centre, #3 y centre, #4 max width, #5 max height.
\newcommand{\asset}[5]{%
  \node[inner sep=0pt,outer sep=0pt] at (#2,#3)
    {\includegraphics[width=#4mm,height=#5mm,keepaspectratio]{assets/#1.png}};}
% Full-width flat band: #1 top, #2 bottom, #3 colour.
\newcommand{\band}[3]{\fill[#3] (0,#1) rectangle (188,#2);}
% Row label at the left of its band: #1 y centre, #2 colour, #3 text.
\newcommand{\rowlabel}[3]{%
  \node[anchor=west,text=#2,font=\fsize{11.5}\bfseries,inner sep=0pt,align=left]
    at (3,#1) {#3};}
% Block arrow pointing down: #1 x centre, #2 y of the tail.
\newcommand{\blockarrow}[2]{%
  \path[fill=ArrowFill,draw=ArrowLine,line width=.35pt,line join=round]
    (#1-1.3,#2) -- (#1+1.3,#2) -- (#1+1.3,#2+2.9) -- (#1+2.9,#2+2.9) --
    (#1,#2+5.6) -- (#1-2.9,#2+2.9) -- (#1-1.3,#2+2.9) -- cycle;}
% Number disc for a design: #1 x, #2 y, #3 digit.
\newcommand{\numdisc}[3]{%
  \path[fill=WarmLabel] (#1,#2) circle[radius=3.5mm];
  \node[text=white,font=\fsize{13}\bfseries,inner sep=0pt] at (#1,#2) {#3};}
% Plain 'Fixed' tag beside a scripted robot: text only, no badge.
\newcommand{\fixedtag}[2]{%
  \node[text=Muted,font=\fsize{6.4}\bfseries,inner sep=0pt] at (#1,#2) {Fixed};}
